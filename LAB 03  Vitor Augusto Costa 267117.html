library(readr)

airlines <- read_csv("airlines.csv")

airports <- read_csv("airports.csv",
  col_select = c(IATA_CODE, CITY, STATE, LATITUDE, LONGITUDE)
)
processa_chunk <- function(df, pos) {
  df %>%
    filter(!str_starts(DESTINATION_AIRPORT, "1")) %>%
    drop_na() %>%
    group_by(DESTINATION_AIRPORT) %>%
    summarise(
      soma_atraso = sum(ARRIVAL_DELAY),
      n_voos = n(),
      .groups = "drop"
    )
}

estatisticas_parciais <- read_csv_chunked(
  file = "flights.csv",
  callback = DataFrameCallback$new(processa_chunk),
  chunk_size = 1000000,
  col_types = cols_only(
    DESTINATION_AIRPORT = col_character(),
    ARRIVAL_DELAY = col_double()
  )
)
flights <- estatisticas_parciais %>%
  group_by(DESTINATION_AIRPORT) %>%
  summarise(
    soma_atraso_total = sum(soma_atraso),
    n_voos_total = sum(n_voos),
    .groups = "drop"
  ) %>%
  mutate(ARRIVAL_DELAY_MEAN = soma_atraso_total / n_voos_total) %>%
  select(DESTINATION_AIRPORT, ARRIVAL_DELAY_MEAN)
aeroportos_ausentes <- anti_join(
  flights, 
  airports, 
  by = c("DESTINATION_AIRPORT" = "IATA_CODE")
)
print(aeroportos_ausentes)
