processa_chunk <- function(df, pos) {
  df %>%
    filter(AIRLINE %in% c("AA", "DL", "UA", "US")) %>%
    mutate(
      DATE = as.Date(paste(YEAR, MONTH, DAY, sep = "-")),
      atrasado = ifelse(ARRIVAL_DELAY > 10, 1, 0)
    ) %>%
    filter(!is.na(ARRIVAL_DELAY)) %>%
    group_by(DATE, AIRLINE) %>%
    summarise(
      voos_atrasados = sum(atrasado),
      voos_totais = n(),
      .groups = "drop"
    )
}
resultados_parciais <- read_csv_chunked(
  file = "flights.csv",
  callback = DataFrameCallback$new(processa_chunk),
  chunk_size = 100000
)

computeStats <- function(df) {
  df %>%
    group_by(MONTH, DAY, AIRLINE) %>%
    summarise(
      soma_atrasados = sum(voos_atrasados),
      soma_totais = sum(voos_totais),
      .groups = "drop"
    ) %>%
    mutate(
      Perc = soma_atrasados / soma_totais,
      Data = as.Date(paste("2015", MONTH, DAY, sep = "-")),
      Cia = AIRLINE
    ) %>%
    select(Cia, Data, Perc)
}

pal <- scale_fill_gradient(low = "#4575b4", high = "#d73027", name = "% Atraso")

baseCalendario <- function(stats, cia) {
  df_cia <- filter(stats, Cia == cia)
  base <- ggcal(df_cia$Data, df_cia$Perc)
  return(base)
}

dados_para_calendario <- computeStats(resultados_agrupados)

cAA <- baseCalendario(dados_para_calendario, "AA")
cDL <- baseCalendario(dados_para_calendario, "DL")
cUA <- baseCalendario(dados_para_calendario, "UA")
cUS <- baseCalendario(dados_para_calendario, "US")
